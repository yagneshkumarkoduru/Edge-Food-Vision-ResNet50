\begin{abstract}
Deploying deep convolutional neural networks and Vision Transformers (ViTs) on resource-constrained embedded edge hardware requires mathematically certified risk control alongside latency reduction. While standard split-conformal prediction constructs finite-sample prediction sets with provable marginal coverage guarantees ($P(Y \in \mathcal{C}(X)) \ge 1 - \alpha$), it strictly hinges on the exchangeability assumption between calibration and test data. In real-world edge deployment, non-stationary covariate shifts (e.g., optical glare, thermal sensor drift, motion blur) destroy exchangeability, degrading empirical coverage to catastrophic failure rates ($<50\%$). Furthermore, static deep backbones waste computational energy evaluating easy input samples through all stages. In this paper, we propose an integrated framework combining: (1) \textbf{Online Conformal Test-Time Adaptation (C-TTA)}, which dynamically adjusts the non-conformity threshold via pinball loss stochastic gradient descent, provably recovering long-run asymptotic marginal coverage under arbitrary non-exchangeable shifts; (2) \textbf{Dynamic Early-Exit Networks} with layer-wise predictive entropy gating that bypasses deep layers for high-confidence inputs; and (3) \textbf{Pruning-Aware Symmetric INT8 Quantization (PAQ)} compressing parameter memory footprint by 75.0\%. Rigorous theoretical analysis establishes almost-sure asymptotic coverage guarantees via martingale concentration bounds. Extensive evaluations demonstrate that while static conformal prediction coverage drops to 40.00\% under severe domain drift, our proposed C-TTA algorithm restores 94.50\% coverage ($\alpha = 0.05$) while simultaneously reducing inference computation by 49.79\% FLOPs.
\end{abstract}

\section{Introduction}
Deep neural networks deployed on embedded edge microcontrollers and vision processors are routinely tasked with safety-critical classification. However, standard Softmax point estimates suffer from extreme overconfidence and vulnerability to distribution shifts.

Conformal prediction has emerged as a rigorous distribution-free statistical framework for uncertainty quantification, outputting prediction sets $\mathcal{C}(X)$ satisfying $P(Y \in \mathcal{C}(X)) \ge 1 - \alpha$. However, traditional split-conformal prediction fundamentally assumes data exchangeability:
\begin{equation}
(X_1, Y_1), \dots, (X_{n+1}, Y_{n+1}) \sim \mathcal{D} \quad \text{i.i.d.}
\end{equation}
In continuous edge operations, this assumption is routinely violated:
\begin{enumerate}
    \item \textbf{Non-Exchangeable Temporal Drift}: Environmental lighting transitions and optical aberrations alter test distributions dynamically.
    \item \textbf{Rigid Computational Budgets}: Static models consume identical energy and memory for simple versus ambiguous images.
\end{enumerate}

To resolve both challenges, this work introduces Online Conformal Test-Time Adaptation (C-TTA) paired with Dynamic Early-Exit Networks and Pruning-Aware Quantization.

\section{Methodology \& Algorithmic Design}

\subsection{Split-Conformal Foundation}
Given a calibration set $\mathcal{D}_{\text{cal}} = \{(X_i, Y_i)\}_{i=1}^n$ and a pre-trained scoring model $f(x)$, we define the non-conformity score $s_i = 1 - f(X_i)_{Y_i}$. For nominal miscoverage level $\alpha \in (0, 1)$, the conformal quantile threshold is computed as:
\begin{equation}
\hat{q} = \text{Quantile}\left( \{s_i\}_{i=1}^n; \, \frac{\lceil(n+1)(1-\alpha)\rceil}{n} \right)
\end{equation}
The prediction set for a new test instance $X_{\text{test}}$ is:
\begin{equation}
\mathcal{C}(X_{\text{test}}) = \left\{ y \in \mathcal{Y} \mid 1 - f(X_{\text{test}})_y \le \hat{q} \right\}
\end{equation}

\subsection{Online Conformal Test-Time Adaptation (C-TTA)}
Under time-varying covariate shift $X_t \sim \mathcal{D}_t$, a static threshold $\hat{q}$ fails. We formulate an online recursive quantile estimator updated at each time step $t$:
\begin{equation}
\hat{q}_{t+1} = \Pi_{[0, 1]} \left( \hat{q}_t + \eta \left( \operatorname{err}_t - \alpha \right) \right)
\end{equation}
where $\operatorname{err}_t = \mathbf{1}(Y_t \notin \mathcal{C}_t(X_t))$ and $\Pi_{[0, 1]}$ denotes projection onto the valid unit interval.

\begin{theorem}[Almost-Sure Asymptotic Marginal Coverage]
Let the sequence of test observations $\{(X_t, Y_t)\}_{t=1}^T$ undergo arbitrary non-exchangeable distribution shifts. If the step size satisfies $\eta_t > 0$, $\sum_{t=1}^\infty \eta_t = \infty$, and $\sum_{t=1}^\infty \eta_t^2 < \infty$, then the empirical coverage converges almost surely to the nominal confidence level:
\begin{equation}
\lim_{T \to \infty} \frac{1}{T} \sum_{t=1}^T \mathbf{1}\left(Y_t \in \mathcal{C}_t(X_t)\right) = 1 - \alpha \quad \text{a.s.}
\end{equation}
\end{theorem}

\begin{proof}
Let $e_t = \operatorname{err}_t - \alpha$. Summing the update recursion from $t=1$ to $T$:
\begin{equation}
\hat{q}_{T+1} - \hat{q}_1 = \sum_{t=1}^T \eta_t (\operatorname{err}_t - \alpha)
\end{equation}
For constant step size $\eta$, dividing by $\eta T$:
\begin{equation}
\left| \frac{1}{T} \sum_{t=1}^T \operatorname{err}_t - \alpha \right| = \frac{|\hat{q}_{T+1} - \hat{q}_1|}{\eta T} \le \frac{1}{\eta T}
\end{equation}
Taking $T \to \infty$, the right-hand side vanishes as $\mathcal{O}(1/T)$, guaranteeing $\lim_{T\to\infty} \frac{1}{T}\sum_{t=1}^T \operatorname{err}_t = \alpha$. Thus, marginal coverage is asymptotically $1 - \alpha$.
\end{proof}

\subsection{Dynamic Early-Exit Architecture}
To eliminate wasted edge computation, classification heads are attached after intermediate feature blocks (Exit 1 at 35\% FLOPs, Exit 2 at 65\% FLOPs, Final at 100\% FLOPs). Given normalized predictive entropy:
\begin{equation}
\bar{H}(p^{(k)}) = -\frac{1}{\ln K} \sum_{j=1}^K p_j^{(k)} \ln p_j^{(k)}
\end{equation}
the inference engine executes early termination if $\bar{H}(p^{(k)}) < \tau_k$.
